using System;
using System.Collections.Generic;
using System.Globalization;
using System.Reflection;
using UnityEditor;
using UnityEngine;

/// <summary>
/// Custom preview window that compensates for Unity's MeshRenderer-only lightmap visualization
/// pipeline. Unity's <c>LightmapPreviewWindow</c> does not include SkinnedMeshRenderers in its
/// per-atlas renderer list (<c>LightmapVisualizationUtility.GetBakedGITextureRenderers</c>) nor
/// in its instanceID-&gt;lightmapIndex reverse table (<c>LightmapEditorSettings.GetLightmapIndex</c>),
/// so even when an SMR is correctly baked it never gets a UV overlay drawn.
///
/// This window:
///   1. Pulls the baked atlas texture for <c>smr.lightmapIndex</c> via reflection on
///      <c>LightmapVisualizationUtility.GetBakedGITexture</c>.
///   2. Walks <c>smr.sharedMesh.GetUVs(uvChannel)</c> + triangle indices and applies
///      <c>uv * lightmapScaleOffset.xy + lightmapScaleOffset.zw</c> to every vertex, then renders
///      the resulting wireframe with GL_LINES on top of the atlas.
///
/// Opened automatically when the user clicks "Open Preview" on a SkinnedMeshRenderer in our
/// custom inspector; can also be opened manually via Tools / Lightmap Probe.
/// </summary>
internal class SkinnedMeshRendererLightmapUVPreviewWindow : EditorWindow
{
    // -------- Reflection cache (LightmapVisualizationUtility internals) --------
    private static MethodInfo s_GetBakedGITextureMethod;
    private static MethodInfo s_GetBakedGITextureHashMethod;
    private static FieldInfo s_VisualisationGITextureField;
    private static FieldInfo s_VisualisationGITextureAvailabilityField;
    private static object s_GITextureTypeBakedValue;
    private static bool s_ReflectionInitialized;

    // -------- Line rendering material --------
    private static Material s_LineMaterial;

    // -------- Window state --------
    [SerializeField] private SkinnedMeshRenderer _target;
    [SerializeField] private int _uvChannel = 1; // Unity convention: channel 1 (mesh.uv2) is the lightmap UV
    [SerializeField] private bool _showWireframe = true;
    [SerializeField] private bool _showBoundingBox = true;
    [SerializeField] private bool _followSelection = true;
    [SerializeField] private Color _wireColor = new Color(1f, 0.85f, 0.2f, 0.95f);
    [SerializeField] private Color _boxColor = new Color(0.2f, 1f, 0.4f, 0.9f);

    // View transform (zoom + pan around the aspect-fit base rect)
    [SerializeField] private float _zoom = 1f;
    [SerializeField] private Vector2 _panPixels = Vector2.zero;
    private const float MinZoom = 0.1f;
    private const float MaxZoom = 32f;

    // Per-target derived data
    private Mesh _cachedMesh;
    private Vector4 _scaleOffset;
    private int _lightmapIndex = -1;
    private Texture2D _atlas;

    // Edge buffers (already transformed into atlas-uv space)
    private Vector2[] _edgeStarts;
    private Vector2[] _edgeEnds;
    private bool _wireframeDirty = true;

    // -------- Static entry points --------

    [MenuItem("Tools/Lightmap Probe/Open SMR UV Preview From Selection")]
    private static void OpenFromSelectionMenu()
    {
        GameObject go = Selection.activeGameObject;
        SkinnedMeshRenderer smr = go != null ? go.GetComponent<SkinnedMeshRenderer>() : null;
        if (smr == null)
        {
            EditorUtility.DisplayDialog(
                "SMR Lightmap UV Preview",
                "Select a GameObject with a SkinnedMeshRenderer first.",
                "OK");
            return;
        }
        Open(smr);
    }

    public static void Open(SkinnedMeshRenderer smr)
    {
        if (smr == null) return;

        InitReflection();

        // Reuse an existing window of this type if it's already open, instead of stacking copies.
        SkinnedMeshRendererLightmapUVPreviewWindow window = GetWindow<SkinnedMeshRendererLightmapUVPreviewWindow>(
            utility: false, title: "SMR Lightmap UV Preview", focus: true);
        window.minSize = new Vector2(420, 420);
        window._target = smr;
        window.RefreshAll();
        window.UpdateTitle();
        window.Show();
    }

    private static void InitReflection()
    {
        if (s_ReflectionInitialized) return;
        s_ReflectionInitialized = true;

        Assembly editorAsm = typeof(Editor).Assembly;

        Type lightmapVisType = editorAsm.GetType("UnityEditor.LightmapVisualizationUtility");
        Type giTextureTypeEnum = editorAsm.GetType("UnityEditor.GITextureType");
        if (giTextureTypeEnum != null && giTextureTypeEnum.IsEnum)
            s_GITextureTypeBakedValue = Enum.Parse(giTextureTypeEnum, "Baked");

        if (lightmapVisType != null && giTextureTypeEnum != null)
        {
            s_GetBakedGITextureMethod = lightmapVisType.GetMethod(
                "GetBakedGITexture",
                BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic,
                null,
                new[] { typeof(int), typeof(int), giTextureTypeEnum },
                null);

            // ShowAtlasGUI calls GetBakedGITextureHash before GetBakedGITexture; the hash call
            // appears to prime the internal async cache, so without it our standalone fetch can
            // keep returning a struct with a null texture indefinitely.
            s_GetBakedGITextureHashMethod = lightmapVisType.GetMethod(
                "GetBakedGITextureHash",
                BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic,
                null,
                new[] { typeof(int), typeof(int), giTextureTypeEnum },
                null);
        }

        Type visType = editorAsm.GetType("UnityEditor.VisualisationGITexture");
        s_VisualisationGITextureField = visType?.GetField("texture", BindingFlags.Public | BindingFlags.Instance);
        s_VisualisationGITextureAvailabilityField = visType?.GetField("textureAvailability", BindingFlags.Public | BindingFlags.Instance);
    }

    // -------- Window lifecycle --------

    private void OnEnable()
    {
        InitReflection();
        Lightmapping.lightingDataUpdated += OnLightingDataUpdated;
        Selection.selectionChanged += OnSelectionChangedExternal;
    }

    private void OnDisable()
    {
        Lightmapping.lightingDataUpdated -= OnLightingDataUpdated;
        Selection.selectionChanged -= OnSelectionChangedExternal;
    }

    private void OnLightingDataUpdated()
    {
        RefreshAll();
        Repaint();
    }

    private void OnSelectionChangedExternal()
    {
        if (!_followSelection) return;
        GameObject go = Selection.activeGameObject;
        if (go == null) return;
        SkinnedMeshRenderer smr = go.GetComponent<SkinnedMeshRenderer>();
        if (smr == null || smr == _target) return;

        _target = smr;
        RefreshAll();
        UpdateTitle();
        Repaint();
    }

    private void UpdateTitle()
    {
        string targetName = _target != null ? _target.name : "(no target)";
        titleContent = new GUIContent("SMR Lightmap UV  -  " + targetName);
    }

    private void RefreshAll()
    {
        _wireframeDirty = true;
        _atlas = null;
        _cachedMesh = null;
        _lightmapIndex = -1;

        if (_target == null) return;

        _cachedMesh = _target.sharedMesh;
        _scaleOffset = _target.lightmapScaleOffset;
        _lightmapIndex = _target.lightmapIndex;
        // First fetch attempt; the texture may still be null if Unity hasn't async-loaded the
        // atlas yet. OnGUI re-tries every frame so it'll show up shortly.
        _atlas = FetchBakedAtlas(_lightmapIndex);
    }

    // Returns the baked atlas texture for the given lightmap index, or null if it isn't loaded.
    //
    // Why this isn't just `LightmapVisualizationUtility.GetBakedGITexture` via reflection: that
    // path was tried first (it's what `LightingSettingsInspector.ShowAtlasGUI` uses for the
    // small inspector thumbnail) but in our standalone EditorWindow context it consistently
    // returns a `VisualisationGITexture` struct whose `texture` field is null - presumably the
    // utility's internal cache only gets populated when the call originates from a renderer
    // inspector or the Lighting window. Going through `LightmapSettings.lightmaps` instead is
    // a stable public API: Bakery (and Unity's Progressive Lightmapper) write the final atlas
    // references there, and they're available immediately with no async-load handshake.
    private static Texture2D FetchBakedAtlas(int lightmapIndex)
    {
        if (lightmapIndex < 0 || lightmapIndex == 65535) return null;

        // Primary path: stable public API, works regardless of editor context.
        var lightmaps = LightmapSettings.lightmaps;
        if (lightmaps != null && lightmapIndex < lightmaps.Length)
        {
            Texture2D tex = lightmaps[lightmapIndex].lightmapColor;
            if (tex != null) return tex;
        }

        // Fallback: original reflection path, in case some editor flow updates the
        // visualization utility before LightmapSettings.lightmaps catches up.
        if (s_GetBakedGITextureMethod == null ||
            s_VisualisationGITextureField == null ||
            s_GITextureTypeBakedValue == null)
        {
            return null;
        }

        if (s_GetBakedGITextureHashMethod != null)
        {
            try
            {
                s_GetBakedGITextureHashMethod.Invoke(
                    null, new object[] { lightmapIndex, 0, s_GITextureTypeBakedValue });
            }
            catch { /* hash priming is best-effort */ }
        }

        object visTex;
        try
        {
            visTex = s_GetBakedGITextureMethod.Invoke(
                null, new object[] { lightmapIndex, 0, s_GITextureTypeBakedValue });
        }
        catch
        {
            return null;
        }

        if (visTex == null) return null;
        return s_VisualisationGITextureField.GetValue(visTex) as Texture2D;
    }

    // -------- UV wireframe build --------

    // Mirrors what Unity's surface shaders do at runtime to compute the lightmap fetch UV:
    //
    //     // HLSL (UnityCG.cginc / Lighting.hlsl):
    //     float2 lightmapUV = v.uv2.xy * unity_LightmapST.xy + unity_LightmapST.zw;
    //
    // where `unity_LightmapST` is just `Renderer.lightmapScaleOffset` packed into a
    // shader uniform, in (scaleX, scaleY, offsetX, offsetY) order. So:
    //
    //     atlasUV.x = meshUV.x * scaleOffset.x + scaleOffset.z
    //     atlasUV.y = meshUV.y * scaleOffset.y + scaleOffset.w
    //
    // We use atlasUV directly as the GL.LINES wireframe coords, with UvToPixel mapping
    // (0,0) -> bottom-left of the displayed atlas rect and (1,1) -> top-right (matches
    // GUI.DrawTexture's natural orientation, which puts texture row 0 at rect.yMax).
    private void BuildWireframe()
    {
        _edgeStarts = null;
        _edgeEnds = null;
        _wireframeDirty = false;
        if (_cachedMesh == null) return;

        List<Vector2> uvs = new List<Vector2>(_cachedMesh.vertexCount);
        _cachedMesh.GetUVs(_uvChannel, uvs);
        if (uvs.Count == 0)
        {
            // Fallback to channel 0 if the user picked an empty channel; lots of skinned
            // meshes ship with only uv0.
            _cachedMesh.GetUVs(0, uvs);
        }
        if (uvs.Count == 0) return;

        Vector2 lmScale = new Vector2(_scaleOffset.x, _scaleOffset.y);
        Vector2 lmOffset = new Vector2(_scaleOffset.z, _scaleOffset.w);

        // Pre-transform every UV once, so the per-edge loop is just three lookups.
        Vector2[] transformedUv = new Vector2[uvs.Count];
        for (int i = 0; i < uvs.Count; i++)
        {
            Vector2 v = uvs[i];
            transformedUv[i] = new Vector2(v.x * lmScale.x + lmOffset.x, v.y * lmScale.y + lmOffset.y);
        }

        // No edge dedup: a duplicate line just overdraws once. Saves a HashSet allocation.
        List<Vector2> startList = new List<Vector2>(2048);
        List<Vector2> endList = new List<Vector2>(2048);

        for (int sm = 0; sm < _cachedMesh.subMeshCount; sm++)
        {
            int[] tris = _cachedMesh.GetTriangles(sm);
            for (int i = 0; i + 2 < tris.Length; i += 3)
            {
                int ai = tris[i], bi = tris[i + 1], ci = tris[i + 2];
                if (ai >= transformedUv.Length || bi >= transformedUv.Length || ci >= transformedUv.Length)
                    continue;
                Vector2 a = transformedUv[ai];
                Vector2 b = transformedUv[bi];
                Vector2 c = transformedUv[ci];

                startList.Add(a); endList.Add(b);
                startList.Add(b); endList.Add(c);
                startList.Add(c); endList.Add(a);
            }
        }

        _edgeStarts = startList.ToArray();
        _edgeEnds = endList.ToArray();
    }

    // -------- IMGUI --------

    private void OnGUI()
    {
        DrawToolbar();

        // Re-fetch atlas every frame. The underlying LightmapVisualizationUtility has its own
        // cache, so this is a cheap dictionary hit when the texture is already loaded; when the
        // bake/load is still pending it gives Unity a chance to hand us a non-null texture once
        // the async load completes.
        if (_target != null && _lightmapIndex >= 0 && _lightmapIndex != 65535)
        {
            _atlas = FetchBakedAtlas(_lightmapIndex);
        }

        // Reserve the rest of the window for the canvas, regardless of event type, so that the
        // layout state is consistent between Layout and Repaint passes.
        Rect canvasRect = GUILayoutUtility.GetRect(
            GUIContent.none, GUIStyle.none,
            GUILayout.ExpandWidth(true),
            GUILayout.ExpandHeight(true));

        DrawCanvas(canvasRect);

        // Keep repainting while the texture isn't available yet so the async load actually
        // surfaces. Without this we'd be stuck displaying "loading" forever until the user
        // moves the mouse over the window.
        if (_target != null && _atlas == null && _lightmapIndex >= 0 && _lightmapIndex != 65535)
        {
            Repaint();
        }
    }

    private void DrawToolbar()
    {
        EditorGUILayout.BeginHorizontal(EditorStyles.toolbar);

        if (GUILayout.Button("Refresh", EditorStyles.toolbarButton, GUILayout.Width(60)))
        {
            RefreshAll();
        }

        if (GUILayout.Button("Pick", EditorStyles.toolbarButton, GUILayout.Width(40)))
        {
            GameObject go = Selection.activeGameObject;
            SkinnedMeshRenderer smr = go != null ? go.GetComponent<SkinnedMeshRenderer>() : null;
            if (smr != null)
            {
                _target = smr;
                RefreshAll();
                UpdateTitle();
            }
        }

        _followSelection = GUILayout.Toggle(
            _followSelection, "Follow Selection", EditorStyles.toolbarButton, GUILayout.Width(110));

        GUILayout.Space(8);

        GUILayout.Label("UV:", GUILayout.Width(24));
        int newChannel = EditorGUILayout.IntPopup(
            _uvChannel,
            new[] { "uv  (0)", "uv2 (1, lightmap)", "uv3 (2)", "uv4 (3)" },
            new[] { 0, 1, 2, 3 },
            EditorStyles.toolbarPopup,
            GUILayout.Width(150));
        if (newChannel != _uvChannel)
        {
            _uvChannel = newChannel;
            _wireframeDirty = true;
        }

        _showWireframe = GUILayout.Toggle(
            _showWireframe, "Wireframe", EditorStyles.toolbarButton, GUILayout.Width(80));
        _showBoundingBox = GUILayout.Toggle(
            _showBoundingBox, "Bounds", EditorStyles.toolbarButton, GUILayout.Width(60));

        if (GUILayout.Button("Reset View", EditorStyles.toolbarButton, GUILayout.Width(80)))
        {
            _zoom = 1f;
            _panPixels = Vector2.zero;
            Repaint();
        }
        GUILayout.Label("Zoom " + _zoom.ToString("0.00") + "x", GUILayout.Width(80));

        GUILayout.FlexibleSpace();

        if (_target != null)
        {
            // Use "R" (round-trip) so the printed numbers match the float values shown in the
            // SkinnedMeshRenderer inspector exactly. With "0.###" they get rounded to 3 decimals
            // which makes them look "different" from the inspector even though the underlying
            // values are identical.
            IFormatProvider fmt = CultureInfo.InvariantCulture.NumberFormat;
            string label = string.Format(fmt,
                "Atlas {0}   |   Tiling ({1:R}, {2:R})   Offset ({3:R}, {4:R})",
                _lightmapIndex,
                _scaleOffset.x, _scaleOffset.y,
                _scaleOffset.z, _scaleOffset.w);
            GUILayout.Label(label);
        }

        EditorGUILayout.EndHorizontal();
    }

    private void DrawCanvas(Rect full)
    {
        if (_target == null)
        {
            DrawCenteredMessage(full,
                "No target.\nUse 'Pick' (with a SkinnedMeshRenderer selected) or\nTools / Lightmap Probe / Open SMR UV Preview From Selection.");
            return;
        }

        if (_atlas == null)
        {
            int lightmapCount = LightmapSettings.lightmaps != null ? LightmapSettings.lightmaps.Length : 0;
            string detail = _lightmapIndex >= lightmapCount
                ? "lightmap index " + _lightmapIndex + " is out of range (LightmapSettings has " + lightmapCount + " atlases)."
                : "atlas " + _lightmapIndex + " has no color texture (not baked yet, or LightingData asset is missing).";
            DrawCenteredMessage(full,
                "No baked atlas available.\n" + detail);
            return;
        }

        // Process zoom/pan input first, so this frame's draw uses the updated transform.
        HandleZoomPanInput(full);

        // Background fills the whole canvas area regardless of where the (zoomed) atlas sits.
        EditorGUI.DrawRect(full, new Color(0.12f, 0.12f, 0.12f, 1f));

        // Aspect-fit base rect, then apply zoom + pan around its center to get the actual rect
        // we draw the atlas (and overlay UVs) into.
        Rect baseFit = AspectFitRect(full, _atlas.width, _atlas.height);
        Rect atlasRect = ApplyZoomPan(baseFit, _zoom, _panPixels);

        // Clip to the canvas area so the atlas doesn't bleed past the toolbar / window edges.
        GUI.BeginClip(full);
        Rect localAtlas = new Rect(atlasRect.x - full.x, atlasRect.y - full.y, atlasRect.width, atlasRect.height);
        if (Event.current.type == EventType.Repaint)
        {
            GUI.DrawTexture(localAtlas, _atlas, ScaleMode.StretchToFill, alphaBlend: false);
        }
        GUI.EndClip();

        if (_wireframeDirty) BuildWireframe();

        if (_showBoundingBox || _showWireframe)
            DrawOverlays(atlasRect, full);
    }

    private static Rect ApplyZoomPan(Rect baseFit, float zoom, Vector2 pan)
    {
        float w = baseFit.width * zoom;
        float h = baseFit.height * zoom;
        float cx = baseFit.center.x + pan.x;
        float cy = baseFit.center.y + pan.y;
        return new Rect(cx - w * 0.5f, cy - h * 0.5f, w, h);
    }

    // Mouse wheel = zoom around cursor (the atlas point under the cursor stays under the cursor
    // before and after the zoom step). Middle / right mouse drag = pan.
    private void HandleZoomPanInput(Rect full)
    {
        Event e = Event.current;
        if (e == null) return;

        if (e.type == EventType.ScrollWheel && full.Contains(e.mousePosition))
        {
            float oldZoom = _zoom;
            // delta.y > 0 means scroll-down on most setups, which we treat as zoom-out.
            float zoomMul = e.delta.y > 0 ? 1f / 1.15f : 1.15f;
            float newZoom = Mathf.Clamp(oldZoom * zoomMul, MinZoom, MaxZoom);
            if (!Mathf.Approximately(newZoom, oldZoom))
            {
                Rect baseFit = AspectFitRect(full, _atlas.width, _atlas.height);
                // Convert "rect center after pan" into a value such that the cursor's atlas-uv
                // doesn't drift. Derivation:
                //   mouse - center = uv_offset_pixels = uvAtMouse * (zoom * baseFit.size)
                //   => uv_offset_pixels scales linearly with zoom
                //   => newCenter = mouse - (mouse - oldCenter) * (newZoom / oldZoom)
                Vector2 oldCenter = baseFit.center + _panPixels;
                Vector2 newCenter = (Vector2)e.mousePosition -
                                    ((Vector2)e.mousePosition - oldCenter) * (newZoom / oldZoom);
                _panPixels = newCenter - baseFit.center;
                _zoom = newZoom;
                Repaint();
            }
            e.Use();
        }

        // Pan: middle-mouse drag, or right-mouse drag (right is more common on Win).
        if (e.type == EventType.MouseDrag && (e.button == 1 || e.button == 2)
            && full.Contains(e.mousePosition))
        {
            _panPixels += e.delta;
            e.Use();
            Repaint();
        }

        // Eat the right-mouse-down so the OS context menu doesn't pop up while dragging.
        if (e.type == EventType.MouseDown && e.button == 1 && full.Contains(e.mousePosition))
        {
            e.Use();
        }
    }

    private void DrawOverlays(Rect canvas, Rect clip)
    {
        if (Event.current.type != EventType.Repaint) return;

        EnsureLineMaterial();
        if (s_LineMaterial == null) return;
        s_LineMaterial.SetPass(0);

        // Use BeginClip so GL.LINES that would land on the toolbar / outside-canvas areas get
        // clipped instead of overdrawing the rest of the window.
        GUI.BeginClip(clip);
        GL.PushMatrix();
        // Pixel matrix is now LOCAL to the clip rect (origin at clip's top-left).
        GL.LoadPixelMatrix(0, clip.width, clip.height, 0);

        Rect localCanvas = new Rect(canvas.x - clip.x, canvas.y - clip.y, canvas.width, canvas.height);

        if (_showWireframe && _edgeStarts != null && _edgeStarts.Length > 0)
        {
            GL.Begin(GL.LINES);
            GL.Color(_wireColor);
            for (int i = 0; i < _edgeStarts.Length; i++)
            {
                Vector2 a = UvToPixel(_edgeStarts[i], localCanvas);
                Vector2 b = UvToPixel(_edgeEnds[i], localCanvas);
                GL.Vertex3(a.x, a.y, 0);
                GL.Vertex3(b.x, b.y, 0);
            }
            GL.End();
        }

        if (_showBoundingBox)
        {
            // The SMR's actual atlas region is just the rect (offset, offset+scale).
            Vector2 min = new Vector2(_scaleOffset.z, _scaleOffset.w);
            Vector2 max = min + new Vector2(_scaleOffset.x, _scaleOffset.y);

            Vector2 a = UvToPixel(new Vector2(min.x, min.y), localCanvas);
            Vector2 b = UvToPixel(new Vector2(max.x, min.y), localCanvas);
            Vector2 c = UvToPixel(new Vector2(max.x, max.y), localCanvas);
            Vector2 d = UvToPixel(new Vector2(min.x, max.y), localCanvas);

            GL.Begin(GL.LINES);
            GL.Color(_boxColor);
            GL.Vertex3(a.x, a.y, 0); GL.Vertex3(b.x, b.y, 0);
            GL.Vertex3(b.x, b.y, 0); GL.Vertex3(c.x, c.y, 0);
            GL.Vertex3(c.x, c.y, 0); GL.Vertex3(d.x, d.y, 0);
            GL.Vertex3(d.x, d.y, 0); GL.Vertex3(a.x, a.y, 0);
            GL.End();
        }

        GL.PopMatrix();
        GUI.EndClip();
    }

    // Map atlas-UV to pixel space within `canvas`.
    //
    // GUI.DrawTexture lays out the texture so that texture row 0 (V=0 in mesh-UV space)
    // appears at the BOTTOM of the destination rect, and the top row at the top. So:
    //   uv (0, 0) -> (canvas.x,           canvas.yMax)   [bottom-left]
    //   uv (1, 1) -> (canvas.xMax,        canvas.y)      [top-right]
    // This matches Unity's convention everywhere else (mesh UVs, lightmapScaleOffset,
    // unity_LightmapST), so wireframe overlays land exactly on the lit charts in the atlas.
    private static Vector2 UvToPixel(Vector2 uv, Rect canvas)
    {
        return new Vector2(
            canvas.x + uv.x * canvas.width,
            canvas.yMax - uv.y * canvas.height);
    }

    private static Rect AspectFitRect(Rect outer, float contentW, float contentH)
    {
        if (contentW <= 0 || contentH <= 0) return outer;
        float scale = Mathf.Min(outer.width / contentW, outer.height / contentH);
        float w = contentW * scale;
        float h = contentH * scale;
        return new Rect(outer.x + (outer.width - w) * 0.5f, outer.y + (outer.height - h) * 0.5f, w, h);
    }

    private static void DrawCenteredMessage(Rect r, string msg)
    {
        var style = new GUIStyle(EditorStyles.centeredGreyMiniLabel) { wordWrap = true };
        GUI.Label(r, msg, style);
    }

    private static void EnsureLineMaterial()
    {
        if (s_LineMaterial != null) return;

        Shader shader = Shader.Find("Hidden/Internal-Colored");
        if (shader == null) return;

        s_LineMaterial = new Material(shader)
        {
            hideFlags = HideFlags.HideAndDontSave
        };
        s_LineMaterial.SetInt("_SrcBlend", (int)UnityEngine.Rendering.BlendMode.SrcAlpha);
        s_LineMaterial.SetInt("_DstBlend", (int)UnityEngine.Rendering.BlendMode.OneMinusSrcAlpha);
        s_LineMaterial.SetInt("_Cull", (int)UnityEngine.Rendering.CullMode.Off);
        s_LineMaterial.SetInt("_ZWrite", 0);
    }
}

using System.Collections.Generic;
using System.Globalization;
using System.Linq;
using System.Reflection;
using UnityEditor;
using UnityEngine;
using UnityEngine.Rendering;

/// <summary>
/// Custom inspector for <see cref="SkinnedMeshRenderer"/> that exposes the same Lightmapping foldout
/// (Scale In Lightmap, Stitch Seams, Lightmap Parameters, baked atlas info) that <see cref="MeshRenderer"/> has.
///
/// Why this is needed:
/// - Unity's built-in <c>SkinnedMeshRendererEditor</c> calls <c>LightingSettingsGUI(false)</c>, which suppresses the
///   Lightmapping section entirely. The reason is a bug in Unity's own code: <c>LightingSettingsInspector
///   .LightmapScaleGUI(meshRenderer: true, ...)</c> performs <c>(MeshRenderer)m_Renderers[0]</c>, which throws
///   <see cref="InvalidCastException"/> when the renderer is a <see cref="SkinnedMeshRenderer"/>. The throw breaks
///   the IMGUI Begin/End foldout pairing and silently hides every section drawn after it (Probes, Additional Settings).
/// - Our project bakes lightmaps with Bakery, which fully supports <see cref="SkinnedMeshRenderer"/>, so we want the
///   editor to expose the lightmap properties.
///
/// Strategy:
/// - Wrap the built-in <c>SkinnedMeshRendererEditor</c> via reflection and reuse the safe parts
///   (<c>OnBlendShapeUI</c>, <c>OnMeshUI</c>, <c>DrawMaterials</c>, <c>OtherSettingsGUI</c>, <c>OnSceneGUI</c>,
///   <c>EditMode</c> button, <c>m_Probes</c>) so we don't have to reimplement them.
/// - Manually draw Lighting + Lightmapping + Probes between Materials and Additional Settings, mirroring the layout
///   of <c>RendererLightingSettings.RenderSettings</c> from Unity 2019.4 source, but using SkinnedMeshRenderer-safe
///   property accessors instead of the buggy <c>LightmapScaleGUI</c> path.
/// </summary>
[CustomEditor(typeof(SkinnedMeshRenderer))]
[CanEditMultipleObjects]
public class SkinnedMeshRendererBakeryInspector : Editor
{

    private void OnEnable()
    {
        InitReflection();

        if (s_DefaultEditorType != null)
            _defaultEditor = CreateEditor(targets, s_DefaultEditorType);

        if (_defaultEditor == null)
            return;

        SerializedObject so = _defaultEditor.serializedObject;
        _aabb = so.FindProperty("m_AABB");
        _dirtyAabb = so.FindProperty("m_DirtyAABB");
        _quality = so.FindProperty("m_Quality");
        _updateWhenOffscreen = so.FindProperty("m_UpdateWhenOffscreen");
        _rootBone = so.FindProperty("m_RootBone");

        _castShadows = so.FindProperty("m_CastShadows");
        _receiveShadows = so.FindProperty("m_ReceiveShadows");
        _receiveGI = so.FindProperty("m_ReceiveGI");
        _lightmapScale = so.FindProperty("m_ScaleInLightmap");
        _stitchLightmapSeams = so.FindProperty("m_StitchLightmapSeams");
        _lightmapParameters = so.FindProperty("m_LightmapParameters");
        // Match what LightingSettingsInspector.ShowAtlasGUI does: read from SerializedProperty
        // rather than the C# accessor, since some Unity builds clamp/convert the runtime value.
        _lightmapIndex = so.FindProperty("m_LightmapIndex");

        _targetGameObjects = targets
            .OfType<SkinnedMeshRenderer>()
            .Select(r => r.gameObject)
            .ToArray();

        _isPrefabAsset = _targetGameObjects.Length > 0 &&
                         PrefabUtility.IsPartOfPrefabAsset(_targetGameObjects[0]);

        Lightmapping.lightingDataUpdated += OnLightingDataUpdated;
    }

    private void OnDisable()
    {
        Lightmapping.lightingDataUpdated -= OnLightingDataUpdated;

        if (_defaultEditor != null)
        {
            DestroyImmediate(_defaultEditor);
            _defaultEditor = null;
        }
    }

    private void OnLightingDataUpdated()
    {
        // Only repaint if the user is actually looking at the Lightmapping section
        // (mirrors what UnityEditor.MeshRendererEditor does internally).
        if (EditorPrefs.GetBool(PrefShowLightmap, true))
            Repaint();
    }

    private static void InitReflection()
    {
        if (s_DefaultEditorType != null)
            return;

        Assembly editorAsm = typeof(Editor).Assembly;

        s_DefaultEditorType = editorAsm.GetType("UnityEditor.SkinnedMeshRendererEditor");
        s_RendererEditorBaseType = editorAsm.GetType("UnityEditor.RendererEditorBase");

        const BindingFlags inst = BindingFlags.Instance | BindingFlags.NonPublic;
        const BindingFlags instAny = BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic;

        if (s_RendererEditorBaseType != null)
        {
            s_DrawMaterialsMethod = s_RendererEditorBaseType.GetMethod(
                "DrawMaterials", inst, null, Type.EmptyTypes, null);

            s_OtherSettingsGUIMethod = s_RendererEditorBaseType.GetMethod(
                "OtherSettingsGUI",
                inst,
                null,
                new[] { typeof(bool), typeof(bool), typeof(bool), typeof(bool) },
                null);

            s_RendererBaseProbesField = s_RendererEditorBaseType.GetField("m_Probes", inst);
            s_RendererBaseLightingField = s_RendererEditorBaseType.GetField("m_Lighting", inst);

            Type probesType = s_RendererEditorBaseType.GetNestedType(
                "Probes", BindingFlags.Public | BindingFlags.NonPublic);
            if (probesType != null)
            {
                s_ProbesOnGUIMethod = probesType.GetMethod(
                    "OnGUI",
                    instAny,
                    null,
                    new[] { typeof(UnityEngine.Object[]), typeof(Renderer), typeof(bool) },
                    null);
            }
        }

        // 2019.4 names the type LightingSettingsInspector; newer Unity versions renamed it to
        // RendererLightingSettings. Probe both so this script works across versions.
        s_LightingInspectorType =
            editorAsm.GetType("UnityEditor.LightingSettingsInspector") ??
            editorAsm.GetType("UnityEditor.RendererLightingSettings");

        if (s_LightingInspectorType != null)
        {
            s_LightmapParametersGUIMethod = s_LightingInspectorType.GetMethod(
                "LightmapParametersGUI",
                BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic,
                null,
                new[] { typeof(SerializedProperty), typeof(GUIContent) },
                null);

            s_ShowAtlasGUIMethod = FindShowAtlasGUI(s_LightingInspectorType);

            s_ShowRealtimeLMGUIMethod = s_LightingInspectorType.GetMethod(
                "ShowRealtimeLMGUI",
                instAny,
                null,
                new[] { typeof(Renderer) },
                null);

            // NOTE: We deliberately do NOT touch LightingSettingsInspector.Styles here. Reading any
            // member of that nested class triggers its cctor, which evaluates
            //   EditorStyles.objectFieldThumb.name + "LightmapPreviewOverlay"
            // EditorStyles isn't initialized during early OnEnable, so the cctor throws NRE,
            // permanently poisoning LightingSettingsInspector.Styles for the entire AppDomain and
            // breaking Unity's own MeshRendererEditor too. OpenPreviewStyle is fetched lazily
            // inside ResolveOpenPreviewStyle() during a real IMGUI event instead.
        }

        Type lightmapVisType = editorAsm.GetType("UnityEditor.LightmapVisualizationUtility");
        Type giTextureTypeEnum = editorAsm.GetType("UnityEditor.GITextureType");
        if (giTextureTypeEnum != null && giTextureTypeEnum.IsEnum)
            s_GITextureTypeBakedValue = Enum.Parse(giTextureTypeEnum, "Baked");

        if (lightmapVisType != null && giTextureTypeEnum != null)
        {
            // STRICT signature match: must be (int, int, GITextureType). A permissive
            // (int, int, <any enum>) match could pick the wrong overload (e.g. one that uses
            // a different enum type) and silently return an empty VisualisationGITexture.
            s_GetBakedGITextureMethod = lightmapVisType.GetMethod(
                "GetBakedGITexture",
                BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic,
                null,
                new[] { typeof(int), typeof(int), giTextureTypeEnum },
                null);
        }

        Type visualisationTextureType = editorAsm.GetType("UnityEditor.VisualisationGITexture");
        s_VisualisationGITextureTextureField = visualisationTextureType?.GetField(
            "texture", BindingFlags.Public | BindingFlags.Instance);

        s_LightmapPreviewWindowType = editorAsm.GetType("UnityEditor.LightmapPreviewWindow");
        if (s_LightmapPreviewWindowType != null)
        {
            s_CreateLightmapPreviewWindowMethod = s_LightmapPreviewWindowType.GetMethod(
                "CreateLightmapPreviewWindow",
                BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic,
                null,
                new[] { typeof(int), typeof(bool), typeof(bool) },
                null);

            s_LightmapPreviewWindow_InstanceIDField = s_LightmapPreviewWindowType.GetField(
                "m_InstanceID", BindingFlags.Instance | BindingFlags.NonPublic);
            s_LightmapPreviewWindow_LightmapIndexField = s_LightmapPreviewWindowType.GetField(
                "m_LightmapIndex", BindingFlags.Instance | BindingFlags.NonPublic);
            s_LightmapPreviewWindow_IsRealtimeField = s_LightmapPreviewWindowType.GetField(
                "m_IsRealtimeLightmap", BindingFlags.Instance | BindingFlags.NonPublic);
        }

        if (s_DefaultEditorType != null)
        {
            s_OnBlendShapeUIMethod = s_DefaultEditorType.GetMethod(
                "OnBlendShapeUI", instAny, null, Type.EmptyTypes, null);
            s_OnMeshUIMethod = s_DefaultEditorType.GetMethod(
                "OnMeshUI", instAny, null, Type.EmptyTypes, null);
            s_DefaultOnSceneGUIMethod = s_DefaultEditorType.GetMethod(
                "OnSceneGUI", instAny, null, Type.EmptyTypes, null);
        }

        Type editModeType = editorAsm.GetType("UnityEditor.EditMode");
        Type sceneViewEditModeEnum = editModeType?.GetNestedType(
            "SceneViewEditMode", BindingFlags.Public | BindingFlags.NonPublic);
        if (sceneViewEditModeEnum != null && sceneViewEditModeEnum.IsEnum)
            s_EditModeColliderEnumValue = Enum.Parse(sceneViewEditModeEnum, "Collider");
        if (editModeType != null && sceneViewEditModeEnum != null)
        {
            s_EditModeDoButtonMethod = editModeType.GetMethod(
                "DoEditModeInspectorModeButton",
                BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic,
                null,
                new[] { sceneViewEditModeEnum, typeof(string), typeof(GUIStyle), typeof(Editor) },
                null);
        }

        Type primitiveBoundsHandleType = editorAsm.GetType("UnityEditor.IMGUI.Controls.PrimitiveBoundsHandle");
        s_PrimitiveBoundsHandleEditModeButtonField = primitiveBoundsHandleType?.GetField(
            "editModeButton",
            BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic);
    }


    public override void OnInspectorGUI()
    {
        if (_defaultEditor == null)
        {
            DrawDefaultInspector();
            return;
        }

        SerializedObject so = _defaultEditor.serializedObject;
        so.Update();

        ComputeContributeGISnapshot();

        DrawEditBoundsButton();

        EditorGUI.BeginChangeCheck();
        EditorGUILayout.PropertyField(_aabb, Styles.Bounds);
        if (EditorGUI.EndChangeCheck())
            _dirtyAabb.boolValue = false;

        s_OnBlendShapeUIMethod?.Invoke(_defaultEditor, null);

        EditorGUILayout.PropertyField(_quality, Styles.Quality);
        EditorGUILayout.PropertyField(_updateWhenOffscreen, Styles.UpdateWhenOffscreen);

        s_OnMeshUIMethod?.Invoke(_defaultEditor, null);

        EditorGUILayout.PropertyField(_rootBone, Styles.RootBone);

        s_DrawMaterialsMethod?.Invoke(_defaultEditor, null);

        DrawLightingFoldout();
        DrawLightmappingFoldout();
        DrawProbesFoldout();
        DrawAdditionalSettings();

        so.ApplyModifiedProperties();
    }

    private void ComputeContributeGISnapshot()
    {
        _frameContributeGI = false;
        _frameMixedContributeGI = false;

        if (_targetGameObjects == null || _targetGameObjects.Length == 0)
            return;

        bool? first = null;
        for (int i = 0; i < _targetGameObjects.Length; i++)
        {
            GameObject go = _targetGameObjects[i];
            if (go == null)
                continue;

            bool has = (GameObjectUtility.GetStaticEditorFlags(go) & StaticEditorFlags.ContributeGI) != 0;
            if (has)
                _frameContributeGI = true;

            if (first == null)
                first = has;
            else if (first.Value != has)
                _frameMixedContributeGI = true;
        }
    }

    private void DrawEditBoundsButton()
    {
        if (s_EditModeDoButtonMethod == null || s_EditModeColliderEnumValue == null)
            return;

        GUIStyle style = (s_PrimitiveBoundsHandleEditModeButtonField?.GetValue(null) as GUIStyle)
                         ?? EditorStyles.miniButton;

        _editModeArgs[0] = s_EditModeColliderEnumValue;
        _editModeArgs[1] = "Edit Bounds";
        _editModeArgs[2] = style;
        _editModeArgs[3] = _defaultEditor;
        s_EditModeDoButtonMethod.Invoke(null, _editModeArgs);
    }

    private void DrawLightingFoldout()
    {
        bool oldShow = EditorPrefs.GetBool(PrefShowLighting, true);
        bool show = EditorGUILayout.BeginFoldoutHeaderGroup(oldShow, Styles.LightingSettings);

        if (show)
        {
            EditorGUI.indentLevel++;

            EditorGUILayout.PropertyField(_castShadows, Styles.CastShadows, true);
            EditorGUILayout.PropertyField(_receiveShadows, Styles.ReceiveShadows, true);

            DrawContributeGIToggle();

            if (_isPrefabAsset)
                EditorGUILayout.HelpBox(Styles.ContributeGIPrefabAssetTip.text, MessageType.Info);

            if (_frameContributeGI && !_frameMixedContributeGI)
            {
                EditorGUI.BeginChangeCheck();
                int newReceive = EditorGUILayout.IntPopup(
                    Styles.ReceiveGITitle,
                    _receiveGI.intValue,
                    Styles.ReceiveGIStrings,
                    Styles.ReceiveGIValues);
                if (EditorGUI.EndChangeCheck())
                    _receiveGI.intValue = newReceive;
            }
            else
            {
                using (new EditorGUI.DisabledScope(true))
                {
                    EditorGUI.showMixedValue = _frameMixedContributeGI;
                    EditorGUILayout.IntPopup(
                        Styles.ReceiveGITitle,
                        (int)ReceiveGI.LightProbes,
                        Styles.ReceiveGIStrings,
                        Styles.ReceiveGIValues);
                    EditorGUI.showMixedValue = false;
                }
            }

            EditorGUI.indentLevel--;
        }

        EditorGUILayout.EndFoldoutHeaderGroup();

        if (show != oldShow)
            EditorPrefs.SetBool(PrefShowLighting, show);
    }

    private void DrawContributeGIToggle()
    {
        // Static flags on a prefab asset can only be modified in Prefab Mode.
        using (new EditorGUI.DisabledScope(_isPrefabAsset))
        {
            EditorGUI.showMixedValue = _frameMixedContributeGI;
            EditorGUI.BeginChangeCheck();
            bool newValue = EditorGUILayout.Toggle(Styles.ContributeGI, _frameContributeGI);
            EditorGUI.showMixedValue = false;

            if (!EditorGUI.EndChangeCheck() || _isPrefabAsset)
                return;

            Undo.RecordObjects(_targetGameObjects, "Set Contribute GI");

            for (int i = 0; i < _targetGameObjects.Length; i++)
            {
                GameObject go = _targetGameObjects[i];
                if (go == null)
                    continue;

                StaticEditorFlags flags = GameObjectUtility.GetStaticEditorFlags(go);
                if (newValue)
                    flags |= StaticEditorFlags.ContributeGI;
                else
                    flags &= ~StaticEditorFlags.ContributeGI;

                GameObjectUtility.SetStaticEditorFlags(go, flags);
                EditorUtility.SetDirty(go);
            }

            // Refresh snapshot so the rest of this frame sees the new value.
            ComputeContributeGISnapshot();
        }
    }

    private void DrawLightmappingFoldout()
    {
        if (!_frameContributeGI || _frameMixedContributeGI)
            return;
        if (_receiveGI.hasMultipleDifferentValues)
            return;
        if ((ReceiveGI)_receiveGI.intValue != ReceiveGI.Lightmaps)
            return;

        bool oldShow = EditorPrefs.GetBool(PrefShowLightmap, true);
        bool show = EditorGUILayout.BeginFoldoutHeaderGroup(oldShow, Styles.LightmapSettings);

        if (show)
        {
            EditorGUI.indentLevel++;

            EditorGUILayout.PropertyField(_lightmapScale, Styles.ScaleInLightmap);
            EditorGUILayout.PropertyField(_stitchLightmapSeams, Styles.StitchLightmapSeams);

            DrawLightmapParameters();
            DrawBakedAndRealtimeLightmap();

            EditorGUI.indentLevel--;
        }

        EditorGUILayout.EndFoldoutHeaderGroup();

        if (show != oldShow)
            EditorPrefs.SetBool(PrefShowLightmap, show);
    }

    private void DrawLightmapParameters()
    {
        if (s_LightmapParametersGUIMethod != null)
        {
            _lightmapParamsArgs[0] = _lightmapParameters;
            _lightmapParamsArgs[1] = Styles.LightmapParameters;
            s_LightmapParametersGUIMethod.Invoke(null, _lightmapParamsArgs);
        }
        else
        {
            EditorGUILayout.PropertyField(_lightmapParameters, Styles.LightmapParameters);
        }
    }

    private void DrawBakedAndRealtimeLightmap()
    {
        object lighting = s_RendererBaseLightingField?.GetValue(_defaultEditor);
        Renderer renderer = target as Renderer;
        bool drewWithBuiltIn = false;

        // Primary path: hand the rendering off to LightingSettingsInspector.ShowAtlasGUI so that
        // the visuals (foldout, thumbnail, atlas overlay style, dev-mode hashes, label formatting)
        // are byte-for-byte identical to MeshRenderer's inspector. We additionally snapshot the
        // currently-open LightmapPreviewWindow set before the call, then after the call detect any
        // window that ShowAtlasGUI's hardcoded "Open Preview" handler may have just popped with
        // CreateLightmapPreviewWindow(instanceID, ..., indexBased: false). For SMR + Bakery that
        // window's UpdateCachedTexture asks LightmapEditorSettings.GetLightmapIndex(instanceID),
        // which Bakery never registers, so the window dead-ends on "texture not available".
        // We close that broken window and re-open it with indexBased: true so it pulls the texture
        // straight from LightmapVisualizationUtility by lightmap index.
        if (s_ShowAtlasGUIMethod != null && lighting != null && renderer != null)
        {
            HashSet<int> windowsBefore = SnapshotLightmapPreviewWindowIds();
            try
            {
                InvokeShowAtlasGUI(lighting, renderer);
                drewWithBuiltIn = true;
            }
            catch (TargetInvocationException)
            {
                // ShowAtlasGUI shouldn't throw on its own, but if some Unity build does, fall through
                // to the manual fallback rather than break the rest of the inspector.
            }

            if (drewWithBuiltIn)
                ReplaceBrokenLightmapPreviewWindows(windowsBefore);
        }

        // Fallback for Unity versions where ShowAtlasGUI isn't reachable via reflection.
        if (!drewWithBuiltIn)
            DrawBakedAtlasGUI();

        // Realtime LM: forward to the original; wrap in try/catch in case some Unity build does an
        // unsafe (MeshRenderer) cast on the renderer internally - we don't want that to break the
        // IMGUI Begin/End pairing for the foldouts that come after.
        if (s_ShowRealtimeLMGUIMethod == null || lighting == null || renderer == null)
            return;

        try
        {
            _realtimeArgs[0] = renderer;
            s_ShowRealtimeLMGUIMethod.Invoke(lighting, _realtimeArgs);
        }
        catch (TargetInvocationException)
        {
            // Swallowed: realtime GI is irrelevant in our Bakery-based pipeline.
        }
    }

    private static MethodInfo FindShowAtlasGUI(Type lightingInspectorType)
    {
        const BindingFlags instAny = BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic;

        // 2019.4: ShowAtlasGUI(int instanceID)
        MethodInfo m = lightingInspectorType.GetMethod(
            "ShowAtlasGUI", instAny, null, new[] { typeof(int) }, null);
        if (m != null)
            return m;

        // 2022+: ShowAtlasGUI(int instanceID, bool showProgressiveSettings, bool isMeshRenderer)
        m = lightingInspectorType.GetMethod(
            "ShowAtlasGUI", instAny, null,
            new[] { typeof(int), typeof(bool), typeof(bool) }, null);
        return m;
    }

    private void InvokeShowAtlasGUI(object lighting, Renderer renderer)
    {
        ParameterInfo[] ps = s_ShowAtlasGUIMethod.GetParameters();
        if (ps.Length == 1)
        {
            _atlasArgs1[0] = renderer.GetInstanceID();
            s_ShowAtlasGUIMethod.Invoke(lighting, _atlasArgs1);
        }
        else if (ps.Length == 3)
        {
            // 2022+: (int instanceID, bool showProgressiveSettings, bool isMeshRenderer)
            _atlasArgs3[0] = renderer.GetInstanceID();
            _atlasArgs3[1] = true;
            _atlasArgs3[2] = false;
            s_ShowAtlasGUIMethod.Invoke(lighting, _atlasArgs3);
        }
    }

    private static HashSet<int> SnapshotLightmapPreviewWindowIds()
    {
        HashSet<int> ids = new HashSet<int>();
        if (s_LightmapPreviewWindowType == null)
            return ids;

        UnityEngine.Object[] windows = Resources.FindObjectsOfTypeAll(s_LightmapPreviewWindowType);
        for (int i = 0; i < windows.Length; i++)
            ids.Add(windows[i].GetInstanceID());
        return ids;
    }

    private void ReplaceBrokenLightmapPreviewWindows(HashSet<int> beforeIds)
    {
        if (s_LightmapPreviewWindowType == null ||
            s_LightmapPreviewWindow_InstanceIDField == null ||
            s_CreateLightmapPreviewWindowMethod == null)
        {
            return;
        }

        UnityEngine.Object[] windows = Resources.FindObjectsOfTypeAll(s_LightmapPreviewWindowType);
        for (int i = 0; i < windows.Length; i++)
        {
            UnityEngine.Object w = windows[i];
            if (beforeIds.Contains(w.GetInstanceID()))
                continue;

            EditorWindow window = w as EditorWindow;
            if (window == null)
                continue;

            // m_InstanceID < 0 means the window was created with indexBased: true already - nothing
            // to fix. We only intercept the instance-based windows (Unity's hardcoded path).
            int targetInstanceID = (int)s_LightmapPreviewWindow_InstanceIDField.GetValue(window);
            if (targetInstanceID < 0)
                continue;

            UnityEngine.Object obj = EditorUtility.InstanceIDToObject(targetInstanceID);
            SkinnedMeshRenderer smr = obj as SkinnedMeshRenderer;
            if (smr == null)
                continue;

            window.Close();

            // Open our custom UV preview window instead of falling back to another
            // LightmapPreviewWindow. The latter would still draw the atlas with an empty UV
            // overlay because Unity's GetBakedGITextureRenderers / GetLightmapIndex tables
            // never contain SkinnedMeshRenderers (verified empirically with SmrLightmapTableProbe).
            // Our window walks the SMR's own mesh.uv2 + lightmapScaleOffset and draws the
            // wireframe on top of the atlas, which is the only way to actually visualize the
            // SMR's UV layout in 2019.4 without patching Unity itself.
            SkinnedMeshRendererLightmapUVPreviewWindow.Open(smr);
        }
    }

    private void DrawBakedAtlasGUI()
    {
        SkinnedMeshRenderer smr = target as SkinnedMeshRenderer;
        if (smr == null)
            return;

        // Read from SerializedProperty (matches what ShowAtlasGUI uses internally) instead of
        // the C# accessor, which can return -1 vs 65535 inconsistently across Unity versions.
        int lightmapIndex = _lightmapIndex != null ? _lightmapIndex.intValue : smr.lightmapIndex;

        // The only definitive "no lightmap" sentinels are < 0 and 65535 (UInt16.MaxValue). Anything
        // else, even unusual values like 65534, is a valid baked state we should attempt to show.
        bool isBaked = lightmapIndex >= 0 && lightmapIndex != 65535;

        // Always render the foldout so the user can see the section exists; show either the data
        // or a diagnostic message inside. This avoids the "everything just silently disappears"
        // failure mode where you can't tell whether the bake or the UI is broken.
        bool oldShow = EditorPrefs.GetBool(PrefShowBakedLM, true);
        bool show = EditorGUILayout.Foldout(oldShow, Styles.BakedLightmap, true);
        if (show != oldShow)
            EditorPrefs.SetBool(PrefShowBakedLM, show);

        if (!show)
            return;

        EditorGUI.indentLevel++;

        if (!isBaked)
        {
            EditorGUILayout.HelpBox(Styles.NotBaked.text, MessageType.Info);
            EditorGUI.indentLevel--;
            GUILayout.Space(5);
            return;
        }

        Texture2D bakedTexture = FetchBakedAtlasTexture(lightmapIndex);
        if (bakedTexture == null)
        {
            EditorGUILayout.HelpBox(Styles.TextureUnavailable.text, MessageType.Warning);

            // Still show the index/tiling so the user knows the bake metadata exists even though
            // the visualization atlas didn't make it into LightmapVisualizationUtility's cache.
            DrawAtlasInfoLabels(lightmapIndex, smr.lightmapScaleOffset);

            EditorGUI.indentLevel--;
            GUILayout.Space(5);
            return;
        }

        GUILayout.BeginHorizontal();
        DrawClickableThumbnail(bakedTexture, lightmapIndex);
        GUILayout.BeginVertical();
        DrawAtlasInfoLabels(lightmapIndex, smr.lightmapScaleOffset);
        GUILayout.EndVertical();
        GUILayout.FlexibleSpace();
        GUILayout.EndHorizontal();

        EditorGUI.indentLevel--;
        GUILayout.Space(5);
    }

    private static void DrawAtlasInfoLabels(int lightmapIndex, Vector4 scaleOffset)
    {
        // Single-line `Label: value` format, matches LightingSettingsInspector.ShowAtlasGUI.
        IFormatProvider fmt = CultureInfo.InvariantCulture.NumberFormat;
        GUILayout.Label(Styles.AtlasIndex.text + ": " + lightmapIndex.ToString(fmt));
        GUILayout.Label(Styles.AtlasTilingX.text + ": " + scaleOffset.x.ToString("0.######", fmt));
        GUILayout.Label(Styles.AtlasTilingY.text + ": " + scaleOffset.y.ToString("0.######", fmt));
        GUILayout.Label(Styles.AtlasOffsetX.text + ": " + scaleOffset.z.ToString("0.######", fmt));
        GUILayout.Label(Styles.AtlasOffsetY.text + ": " + scaleOffset.w.ToString("0.######", fmt));
    }

    private Texture2D FetchBakedAtlasTexture(int lightmapIndex)
    {
        if (s_GetBakedGITextureMethod == null || s_VisualisationGITextureTextureField == null ||
            s_GITextureTypeBakedValue == null)
        {
            return null;
        }

        _getBakedTexArgs[0] = lightmapIndex;
        _getBakedTexArgs[1] = 0;
        _getBakedTexArgs[2] = s_GITextureTypeBakedValue;

        object visualisationTexture = s_GetBakedGITextureMethod.Invoke(null, _getBakedTexArgs);
        if (visualisationTexture == null)
            return null;

        return s_VisualisationGITextureTextureField.GetValue(visualisationTexture) as Texture2D;
    }

    private void DrawClickableThumbnail(Texture2D texture, int lightmapIndex)
    {
        const int previewPadding = 30;
        const int previewWidth = 100; // 104 - 4 padding, matches LightingSettingsInspector

        GUILayout.Space(previewPadding);

        Rect rect = GUILayoutUtility.GetRect(previewWidth, previewWidth, EditorStyles.objectField);
        Rect buttonRect = new Rect(rect.xMax - 70, rect.yMax - 14, 70, 14);

        if (Event.current.type == EventType.MouseDown)
        {
            bool clickedButton = buttonRect.Contains(Event.current.mousePosition) &&
                                 Event.current.clickCount == 1;
            bool doubleClickedThumb = rect.Contains(Event.current.mousePosition) &&
                                      Event.current.clickCount == 2;

            if (clickedButton || doubleClickedThumb)
            {
                OpenLightmapPreviewByIndex(lightmapIndex);
                Event.current.Use();
            }
        }

        EditorGUI.Toggle(rect, false, EditorStyles.objectFieldThumb);

        if (Event.current.type == EventType.Repaint)
        {
            Rect inner = EditorStyles.objectFieldThumb.padding.Remove(rect);
            EditorGUI.DrawPreviewTexture(inner, texture);

            GUIStyle overlay = ResolveOpenPreviewStyle() ?? EditorStyles.miniButton;
            overlay.Draw(rect, Styles.OpenPreview, false, false, false, false);
        }

        float spacing = Mathf.Max(5f, EditorGUIUtility.labelWidth - previewPadding - (previewWidth + 4));
        GUILayout.Space(spacing);
    }

    private void OpenLightmapPreviewByIndex(int lightmapIndex, bool realtime = false)
    {
        // For our SMR target, prefer our custom UV-overlay window. We only fall back to Unity's
        // LightmapPreviewWindow when something has gone wrong (no SMR target at all).
        SkinnedMeshRenderer smr = target as SkinnedMeshRenderer;
        if (smr != null)
        {
            SkinnedMeshRendererLightmapUVPreviewWindow.Open(smr);
            return;
        }

        if (s_CreateLightmapPreviewWindowMethod == null)
            return;

        // The third parameter is `indexBased`; passing true makes LightmapPreviewWindow skip the
        // LightmapEditorSettings.GetLightmapIndex(instanceID) reverse lookup that fails for SMR + Bakery.
        _openPreviewArgs[0] = lightmapIndex;
        _openPreviewArgs[1] = realtime;
        _openPreviewArgs[2] = true;
        s_CreateLightmapPreviewWindowMethod.Invoke(null, _openPreviewArgs);
    }

    // Lazy-resolves LightingSettingsInspector.Styles.OpenPreviewStyle. Must NEVER be called from
    // OnEnable / static ctors / CompilationPipeline events: that nested Styles class's cctor reads
    // EditorStyles.objectFieldThumb.name, which throws NRE before EditorStyles is ready, and the
    // resulting TypeInitializationException permanently poisons the type for the entire AppDomain
    // (also breaking Unity's own MeshRendererEditor). Calling this from inside an IMGUI event is
    // safe because EditorStyles is guaranteed to be initialized by then.
    private static GUIStyle ResolveOpenPreviewStyle()
    {
        if (s_OpenPreviewStyleResolved)
            return s_OpenPreviewStyle;

        // Defensive: only attempt the lookup during a live IMGUI event.
        if (Event.current == null)
            return null;

        s_OpenPreviewStyleResolved = true;

        try
        {
            Type stylesType = s_LightingInspectorType?.GetNestedType(
                "Styles", BindingFlags.Public | BindingFlags.NonPublic);
            FieldInfo openPreviewStyleField = stylesType?.GetField(
                "OpenPreviewStyle",
                BindingFlags.Public | BindingFlags.NonPublic | BindingFlags.Static);
            s_OpenPreviewStyle = openPreviewStyleField?.GetValue(null) as GUIStyle;
        }
        catch (TypeInitializationException)
        {
            // Already poisoned (probably by an earlier domain reload); silently fall back.
            s_OpenPreviewStyle = null;
        }
        catch (Exception)
        {
            s_OpenPreviewStyle = null;
        }

        return s_OpenPreviewStyle;
    }

    private void DrawProbesFoldout()
    {
        bool oldShow = EditorPrefs.GetBool(PrefShowProbes, true);
        bool show = EditorGUILayout.BeginFoldoutHeaderGroup(oldShow, Styles.ProbeSettings);

        if (show)
        {
            EditorGUI.indentLevel++;

            if (s_RendererBaseProbesField != null && s_ProbesOnGUIMethod != null)
            {
                object probes = s_RendererBaseProbesField.GetValue(_defaultEditor);
                Renderer renderer = target as Renderer;
                if (probes != null && renderer != null)
                {
                    _probesArgs[0] = targets;
                    _probesArgs[1] = renderer;
                    _probesArgs[2] = false;
                    s_ProbesOnGUIMethod.Invoke(probes, _probesArgs);
                }
            }

            EditorGUI.indentLevel--;
        }

        EditorGUILayout.EndFoldoutHeaderGroup();

        if (show != oldShow)
            EditorPrefs.SetBool(PrefShowProbes, show);
    }

        s_OtherSettingsGUIMethod.Invoke(
            _defaultEditor, new object[] { false, true, false, false });
    }

using System;
using System.Collections.Generic;
using System.IO;
using System.Linq;
using System.Reflection;
using System.Text;
using UnityEditor;
using UnityEngine;

/// <summary>
/// One-shot diagnostic. Verifies the hypothesis that Unity's per-renderer "instanceID -> lightmapIndex"
/// reverse-lookup table (used by <c>LightmapPreviewWindow.UpdateCachedTexture</c>) and the
/// "lightmapIndex -> renderers" table (used by <c>LightmapVisualizationUtility</c> for the UV overlay)
/// only contain <see cref="MeshRenderer"/> entries even when SkinnedMeshRenderers in the scene have a
/// valid <c>renderer.lightmapIndex</c> set by Bakery.
///
/// Run via:
///   Tools > Lightmap Probe > Dump Renderer-Atlas Tables
///
/// Delete this file when you're done diagnosing.
/// </summary>
internal static class SmrLightmapTableProbe
{
    private const string MenuRoot = "Tools/Lightmap Probe/";

    [MenuItem(MenuRoot + "Dump Renderer-Atlas Tables")]
    private static void DumpRendererAtlasTables()
    {
        if (!ResolveReflection(out var ctx))
        {
            Debug.LogError("[SMR Probe] Failed to resolve required Unity internals via reflection.");
            return;
        }

        // 1. Collect every renderer in the scene that has a (presumably) baked lightmap.
        // Note: 2019.4 has no `includeInactive` overload; the inactive ones aren't critical here
        // since their lightmap state is rarely set anyway.
        Renderer[] allRenderers = UnityEngine.Object.FindObjectsOfType<Renderer>();

        List<RendererInfo> baked = new List<RendererInfo>();
        foreach (Renderer r in allRenderers)
        {
            if (r is MeshRenderer || r is SkinnedMeshRenderer)
            {
                int idx = r.lightmapIndex;
                if (idx >= 0 && idx != 65535)
                    baked.Add(new RendererInfo { renderer = r, lightmapIndex = idx });
            }
        }

        if (baked.Count == 0)
        {
            Debug.LogWarning("[SMR Probe] No renderers with a baked lightmap found in the active scene.");
            return;
        }

        // 2. For each baked renderer, ask LightmapEditorSettings.GetLightmapIndex(instanceID, out idx).
        //    This is the reverse table LightmapPreviewWindow needs for instance-based preview.
        // SkinnedMeshRenderers come FIRST so they're visible if console truncates.
        var rendererPriority = new Func<RendererInfo, int>(b => b.renderer is SkinnedMeshRenderer ? 0 : 1);
        var orderedBaked = baked.OrderBy(rendererPriority)
                                .ThenBy(b => b.lightmapIndex)
                                .ThenBy(b => b.renderer.name)
                                .ToArray();

        // First pass: counts (so we can print summary BEFORE the table - it's the actual answer)
        int mrInTable = 0, mrTotal = 0, smrInTable = 0, smrTotal = 0;
        var perRendererState = new List<(RendererInfo info, bool found, int reverseIdx)>(orderedBaked.Length);

        object[] reverseArgs = new object[2];
        foreach (RendererInfo info in orderedBaked)
        {
            reverseArgs[0] = info.renderer.GetInstanceID();
            reverseArgs[1] = 0;
            bool found = (bool)ctx.GetLightmapIndexMethod.Invoke(null, reverseArgs);
            int reverseIdx = (int)reverseArgs[1];
            perRendererState.Add((info, found, reverseIdx));

            bool isMR = info.renderer is MeshRenderer;
            if (isMR) { mrTotal++; if (found) mrInTable++; }
            else      { smrTotal++; if (found) smrInTable++; }
        }

        // Summary FIRST, then details. This way even if console truncates, the answer is at the top.
        StringBuilder sb = new StringBuilder();
        sb.AppendLine("=== SUMMARY (the actual answer) ===");
        sb.AppendLine($"  MR  in reverse table: {mrInTable} / {mrTotal}");
        sb.AppendLine($"  SMR in reverse table: {smrInTable} / {smrTotal}");
        sb.AppendLine();

        // 3. Per-atlas object lists (also a small section, prints first).
        IEnumerable<int> uniqueIndices = baked.Select(b => b.lightmapIndex).Distinct().OrderBy(i => i);
        object[] atlasArgs = new object[1];

        sb.AppendLine("=== LightmapVisualizationUtility.GetBakedGITextureRenderers(lightmapIndex) ===");
        foreach (int lmIdx in uniqueIndices)
        {
            atlasArgs[0] = lmIdx;
            object result = ctx.GetBakedGITextureRenderersMethod.Invoke(null, atlasArgs);
            GameObject[] gos = result as GameObject[];

            int gosLen = gos?.Length ?? 0;
            var inThisAtlas = baked.Where(b => b.lightmapIndex == lmIdx).ToArray();
            int sceneMR = inThisAtlas.Count(b => b.renderer is MeshRenderer);
            int sceneSMR = inThisAtlas.Count(b => b.renderer is SkinnedMeshRenderer);

            int mrInList = 0, smrInList = 0, otherInList = 0;
            if (gos != null)
            {
                foreach (var go in gos)
                {
                    if (go == null) { otherInList++; continue; }
                    if (go.GetComponent<MeshRenderer>() != null) mrInList++;
                    else if (go.GetComponent<SkinnedMeshRenderer>() != null) smrInList++;
                    else otherInList++;
                }
            }

            sb.AppendLine($"Atlas {lmIdx,2}: returned {gosLen,3} | scene has {sceneMR} MR + {sceneSMR} SMR | returned has {mrInList} MR + {smrInList} SMR + {otherInList} other/null");
        }
        sb.AppendLine();

        // 4. Detailed renderer-by-renderer table (SMR first).
        sb.AppendLine("=== LightmapEditorSettings.GetLightmapIndex(instanceID) [SMR FIRST] ===");
        sb.AppendLine($"{"Type",-22} {"LM idx (field)",-15} {"Reverse lookup",-20} {"Name"}");
        sb.AppendLine(new string('-', 110));

        foreach (var (info, found, reverseIdx) in perRendererState)
        {
            string state = found ? $"FOUND -> {reverseIdx}" : "NOT IN TABLE";
            sb.AppendLine($"{info.renderer.GetType().Name,-22} {info.lightmapIndex,-15} {state,-20} {info.renderer.name}");
        }

        // 5. Per-atlas object list (full GameObject names) - tail of report.
        sb.AppendLine();
        sb.AppendLine("=== Per-atlas object lists (full names) ===");
        foreach (int lmIdx in uniqueIndices)
        {
            atlasArgs[0] = lmIdx;
            object result = ctx.GetBakedGITextureRenderersMethod.Invoke(null, atlasArgs);
            GameObject[] gos = result as GameObject[];

            sb.AppendLine();
            sb.AppendLine($"Atlas {lmIdx}: {(gos?.Length ?? 0)} entries");
            if (gos != null)
            {
                for (int i = 0; i < gos.Length; i++)
                {
                    GameObject go = gos[i];
                    string compTag = go == null
                        ? "<null>"
                        : (go.GetComponent<MeshRenderer>() != null ? "MR"
                          : go.GetComponent<SkinnedMeshRenderer>() != null ? "SMR"
                          : "?");
                    sb.AppendLine($"    [{i,3}] ({compTag,-3}) {(go != null ? go.name : "")}");
                }
            }
        }

        // Print the SUMMARY portion to console (always visible), and dump the full report to a file.
        // Console gets truncated for very long strings; the file is the source of truth.
        string fullReport = sb.ToString();

        string outputPath = Path.Combine(Application.dataPath, "..", "SmrLightmapTableProbe.txt");
        outputPath = Path.GetFullPath(outputPath);
        try
        {
            File.WriteAllText(outputPath, fullReport);
            Debug.Log($"[SMR Probe] Full report written to:\n{outputPath}\n\n--- SUMMARY ---\n" +
                      $"MR  in reverse table: {mrInTable}/{mrTotal}\n" +
                      $"SMR in reverse table: {smrInTable}/{smrTotal}");
        }
        catch (Exception ex)
        {
            Debug.LogError($"[SMR Probe] Failed to write report file: {ex.Message}");
            Debug.Log(fullReport);
        }
    }

    private struct RendererInfo
    {
        public Renderer renderer;
        public int lightmapIndex;
    }

    private struct ReflectionContext
    {
        public MethodInfo GetLightmapIndexMethod;          // LightmapEditorSettings.GetLightmapIndex(int, out int)
        public MethodInfo GetBakedGITextureRenderersMethod; // LightmapVisualizationUtility.GetBakedGITextureRenderers(int) -> GameObject[]
    }

    private static bool ResolveReflection(out ReflectionContext ctx)
    {
        ctx = default;

        Assembly editorAsm = typeof(Editor).Assembly;

        Type lightmapEditorSettings = typeof(LightmapEditorSettings);
        ctx.GetLightmapIndexMethod = lightmapEditorSettings.GetMethod(
            "GetLightmapIndex",
            BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic,
            null,
            new[] { typeof(int), typeof(int).MakeByRefType() },
            null);

        Type lightmapVis = editorAsm.GetType("UnityEditor.LightmapVisualizationUtility");
        ctx.GetBakedGITextureRenderersMethod = lightmapVis?.GetMethod(
            "GetBakedGITextureRenderers",
            BindingFlags.Static | BindingFlags.Public | BindingFlags.NonPublic,
            null,
            new[] { typeof(int) },
            null);

        if (ctx.GetLightmapIndexMethod == null)
            Debug.LogError("[SMR Probe] LightmapEditorSettings.GetLightmapIndex(int, out int) not found.");
        if (ctx.GetBakedGITextureRenderersMethod == null)
            Debug.LogError("[SMR Probe] LightmapVisualizationUtility.GetBakedGITextureRenderers(int) not found.");

        return ctx.GetLightmapIndexMethod != null && ctx.GetBakedGITextureRenderersMethod != null;
    }
